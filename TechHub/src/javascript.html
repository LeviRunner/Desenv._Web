📦 Tipos de Dados em JavaScript

O que são Tipos de Dados?

JavaScript é uma linguagem dinamicamente tipada, o que significa que as variáveis podem conter qualquer tipo de dado e o tipo pode mudar durante a execução.

Tipos Primitivos

String (Texto)


let nome = "João Silva";
let mensagem = 'Olá, mundo!';
let template = `Bem-vindo, ${nome}!`;

Number (Número)


let idade = 25;
let preco = 19.99;
let infinito = Infinity;
let naoEhNumero = NaN; // Not a Number

Boolean (Verdadeiro/Falso)


let ativo = true;
let desativado = false;
let resultado = 10 > 5; // true

Null e Undefined


let vazio = null; // Valor ausente intencionalmente
let indefinido; // undefined - variável declarada mas não inicializada

Tipos Complexos

Array (Lista)


let frutas = ["maçã", "banana", "laranja"];
let numeros = [1, 2, 3, 4, 5];
let misto = [1, "texto", true, null];

// Acessar elementos
console.log(frutas[0]); // "maçã"
console.log(frutas.length); // 3

Object (Objeto)


let pessoa = {
    nome: "Maria",
    idade: 30,
    profissao: "Desenvolvedora",
    ativo: true
};

// Acessar propriedades
console.log(pessoa.nome); // "Maria"
console.log(pessoa["idade"]); // 30

➕ Operadores JavaScript

Operadores Aritméticos


let a = 10;
let b = 3;

console.log(a + b);  // 13 (adição)
console.log(a - b);  // 7 (subtração)
console.log(a * b);  // 30 (multiplicação)
console.log(a / b);  // 3.333... (divisão)
console.log(a % b);  // 1 (resto da divisão)
console.log(a ** b); // 1000 (exponenciação)

Operadores de Comparação


let x = 10;
let y = "10";

console.log(x == y);   // true (comparação de valor)
console.log(x === y);  // false (comparação de valor e tipo)
console.log(x != y);   // false
console.log(x !== y);  // true
console.log(x > 5);    // true
console.log(x <= 10);  // true

Operadores Lógicos


let a = true;
let b = false;

console.log(a && b);  // false (E lógico)
console.log(a || b);  // true (OU lógico)
console.log(!a);      // false (NÃO lógico)

// Uso prático
if (idade >= 18 && temCarteira) {
    console.log("Pode dirigir");
}

Operador Ternário (Condicional)


let idade = 20;
let status = idade >= 18 ? "Maior de idade" : "Menor de idade";
console.log(status); // "Maior de idade"

// Sintaxe: condição ? valorSim : valorNao

🎮 Estruturas de Controle

Condicional IF


let nota = 85;

if (nota >= 90) {
    console.log("Excelente!");
} else if (nota >= 80) {
    console.log("Bom!");
} else if (nota >= 70) {
    console.log("Satisfatório");
} else {
    console.log("Insuficiente");
}

Switch Case


let diaSemana = 3;
let nomeDia;

switch(diaSemana) {
    case 1:
        nomeDia = "Segunda";
        break;
    case 2:
        nomeDia = "Terça";
        break;
    case 3:
        nomeDia = "Quarta";
        break;
    default:
        nomeDia = "Dia inválido";
}

console.log(nomeDia); // "Quarta"

Loops - FOR


// Loop tradicional
for (let i = 0; i < 5; i++) {
    console.log(i); // 0, 1, 2, 3, 4
}

// For...of (para valores)
let numeros = [10, 20, 30];
for (let num of numeros) {
    console.log(num); // 10, 20, 30
}

// For...in (para índices/chaves)
for (let idx in numeros) {
    console.log(idx); // 0, 1, 2
}

Loops - WHILE


let contador = 0;

while (contador < 5) {
    console.log(contador);
    contador++;
}

// Do...While (executa pelo menos uma vez)
let x = 10;
do {
    console.log(x);
    x++;
} while (x < 5); // Executa uma vez mesmo que a condição seja falsa

🔧 Funções em JavaScript

Função Tradicional


// Declaração de função
function saudacao(nome) {
    return `Olá, ${nome}!`;
}

console.log(saudacao("João")); // "Olá, João!"

// Função com múltiplos parâmetros
function somar(a, b) {
    return a + b;
}

console.log(somar(5, 3)); // 8

Função Anônima e Arrow Function


// Função anônima
const quadrado = function(num) {
    return num * num;
};

console.log(quadrado(5)); // 25

// Arrow function (sintaxe moderna)
const cubo = (num) => {
    return num * num * num;
};

// Arrow function simplificada
const dobro = (num) => num * 2;

console.log(dobro(10)); // 20

Parâmetros Padrão e Rest


// Parâmetro padrão
function cumprimento(nome = "Visitante") {
    console.log(`Bem-vindo, ${nome}!`);
}

cumprimento(); // "Bem-vindo, Visitante!"
cumprimento("Ana"); // "Bem-vindo, Ana!"

// Rest parameters (aceita múltiplos argumentos)
function soma(...numeros) {
    let total = 0;
    for (let num of numeros) {
        total += num;
    }
    return total;
}

console.log(soma(1, 2, 3, 4, 5)); // 15

Escopo de Variáveis


var global = "Escopo global";

function exemplo() {
    let local = "Escopo local";
    const constante = "Não pode mudar";
    
    console.log(global); // "Escopo global" ✓
    console.log(local);  // "Escopo local" ✓
}

console.log(local); // ReferenceError - não existe aqui

📋 Métodos de Array

Métodos Principais


let numeros = [1, 2, 3, 4, 5];

// map() - transforma cada elemento
let dobrados = numeros.map(n => n * 2);
console.log(dobrados); // [2, 4, 6, 8, 10]

// filter() - filtra elementos
let pares = numeros.filter(n => n % 2 === 0);
console.log(pares); // [2, 4]

// reduce() - reduz a um único valor
let soma = numeros.reduce((acc, n) => acc + n, 0);
console.log(soma); // 15

// find() - encontra primeiro elemento
let primeiroPar = numeros.find(n => n % 2 === 0);
console.log(primeiroPar); // 2

// forEach() - itera sobre elementos
numeros.forEach((n, idx) => {
    console.log(`Índice ${idx}: ${n}`);
});

Manipulação de Arrays


let frutas = ["maçã", "banana"];

// Adicionar elementos
frutas.push("laranja"); // Final
frutas.unshift("morango"); // Início

// Remover elementos
frutas.pop(); // Final
frutas.shift(); // Início

// Verificar inclusão
console.log(frutas.includes("banana")); // true

// Índice de elemento
console.log(frutas.indexOf("maçã")); // índice

// Concatenar arrays
let lista1 = [1, 2];
let lista2 = [3, 4];
let completa = lista1.concat(lista2);
console.log(completa); // [1, 2, 3, 4]

// Ordenar
let numeros = [3, 1, 4, 1, 5];
numeros.sort((a, b) => a - b);
console.log(numeros); // [1, 1, 3, 4, 5]

📦 Trabalhar com Objetos

Criando e Acessando Objetos


// Objeto literal
let livro = {
    titulo: "JavaScript Moderno",
    autor: "Jon Doe",
    paginas: 450,
    disponivel: true,
    
    // Método
    descricao() {
        return `${this.titulo} por ${this.autor}`;
    }
};

// Acessar propriedades
console.log(livro.titulo); // "JavaScript Moderno"
console.log(livro["autor"]); // "Jon Doe"
console.log(livro.descricao()); // "JavaScript Moderno por Jon Doe"

// Modificar/adicionar propriedades
livro.ano = 2024;
livro.paginas = 500;

// Deletar propriedade
delete livro.disponivel;

Iterando sobre Objetos


let pessoa = {
    nome: "Carlos",
    idade: 28,
    profissao: "Programador"
};

// for...in (itera chaves)
for (let chave in pessoa) {
    console.log(`${chave}: ${pessoa[chave]}`);
}

// Object.keys() - obter todas as chaves
let chaves = Object.keys(pessoa);
console.log(chaves); // ["nome", "idade", "profissao"]

// Object.values() - obter todos os valores
let valores = Object.values(pessoa);
console.log(valores); // ["Carlos", 28, "Programador"]

// Object.entries() - pares chave-valor
let entradas = Object.entries(pessoa);
for (let [chave, valor] of entradas) {
    console.log(`${chave}: ${valor}`);
}

1. Use let e const em vez de var

3. Tratamento de Erros


try {
    let resultado = JSON.parse('JSON inválido');
} catch (erro) {
    console.error("Erro ao fazer parse:", erro.message);
} finally {
    console.log("Limpeza realizada");
}

// Validação de entrada
function dividir(a, b) {
    if (b === 0) {
        throw new Error("Divisão por zero não permitida!");
    }
    return a / b;
}

4. Documentação e Comentários


/**
 * Calcula o preço final com desconto
 * @param {number} preco - Preço original
 * @param {number} desconto - Percentual de desconto
 * @returns {number} Preço final
 */
function calcularPrecoComDesconto(preco, desconto) {
    // Validar entrada
    if (preco < 0 || desconto < 0) {
        return 0;
    }
    return preco * (1 - desconto / 100);
}

🎯 Exemplo Prático: Calculadora Simples

HTML + JavaScript


// HTML
<input type="number" id="num1" placeholder="Número 1">
<input type="number" id="num2" placeholder="Número 2">
<button id="btnSomar">+</button>
<button id="btnSubtrair">-</button>
<div id="resultado"></div>

// JavaScript
const num1 = document.getElementById("num1");
const num2 = document.getElementById("num2");
const btnSomar = document.getElementById("btnSomar");
const btnSubtrair = document.getElementById("btnSubtrair");
const resultado = document.getElementById("resultado");

function exibirResultado(valor) {
    resultado.textContent = `Resultado: ${valor}`;
}

btnSomar.addEventListener("click", () => {
    let r = parseFloat(num1.value) + parseFloat(num2.value);
    exibirResultado(r);
});

btnSubtrair.addEventListener("click", () => {
    let r = parseFloat(num1.value) - parseFloat(num2.value);
    exibirResultado(r);
});
